show databases

// creates if not exists
use practise 

// create collection
db.createCollection("students")

//see all collections
show collections

//insert document
db.students.insertOne({
  name: "Ravi",
  age: 21,
  city: "Hyderabad"
})

//insert many documents
db.students.insertMany([
  { name: "Rahul", age: 22, city: "Chennai" },
  { name: "Mohan", age: 19, city: "Mumbai" },
  { name: "Arun", age: 23, city: "Delhi" }
])

// find one document
db.students.findOne()

// find many or all 
db.students.find()

// equals operator 
db.students.find({ age: { $eq: 21 } })

// greater-than operator
db.students.find({ age: { $gt: 20 } })

// less-than operator
db.students.find({ age: { $lt: 22 } })

//not equals operator
db.students.find({ city: { $ne: "Delhi" } })

//in operator
db.students.find({ city: { $in: ["Delhi", "Mumbai"] } })

// AND operator
db.students.find({ $and: [{ age: { $gt: 20 } },{ city: "Hyderabad" }]})
    
// OR operator
db.students.find({$or: [{ age: 19 },{ age: 23 }]})

// update one document
db.students.updateOne({ name: "Ravi" },{ $set: { age: 25 } })

// update many document
db.students.updateMany({ city: "Mumbai" },{ $set: { city: "Pune" }})

// delete document
db.students.deleteOne({ name: "Arun" })

//delete many document
db.students.deleteOne({ name: "Arun" })

//update one record use (push)
db.students.updateOne(
  { name: "Rahul" },
  { $push: { skills: "NodeJS" } }
)

db.students.updateOne(
  { name: "rahul" },
  { $push: { skills: { $each: ["React", "Express"] } } }
)


//pop
db.students.updateOne(
  { name: "Rahul" },
  { $pop: { skills: 1 } }
)

//pull
db.students.updateOne(
  { name: "rahul" },
  { $pullAll: { skills: ["React", "Express"] } }
)
